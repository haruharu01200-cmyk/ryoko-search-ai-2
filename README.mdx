'use client';

import { useState } from 'react';

type Message = {
  sender: 'user' | 'ai';
  text: string;
};

export default function TravelSearchAI() {
  const [input, setInput] = useState('');
  const [messages, setMessages] = useState<Message[]>([
    { sender: 'ai', text: 'こんにちは！旅行検索AIです。行きたい場所や希望の条件（温泉、北海道など）を教えてください！' }
  ]);
  const [loading, setLoading] = useState(false);

  // 正しいApplication IDに変更しました
  const RAKUTEN_APPLICATION_ID = 'b7730ddc-9bb7-4503-acb0-04b2efabb8c7';

  const handleSend = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!input.trim() || loading) return;

    const userMessage = input;
    setInput('');
    setMessages((prev) => [...prev, { sender: 'user', text: userMessage }]);
    setLoading(true);

    try {
      const url = `https://app.rakuten.co.jp/services/api/Travel/KeywordHotelSearch/20131024?format=json&keyword=${encodeURIComponent(
        userMessage
      )}&applicationId=${RAKUTEN_APPLICATION_ID}`;

      const response = await fetch(url);
      const data = await response.json();

      if (data.hotels && data.hotels.length > 0) {
        const hotelName = data.hotels[0].hotel[0].hotelBasicInfo.hotelName;
        const hotelUrl = data.hotels[0].hotel[0].hotelBasicInfo.hotelInformationUrl;

        setMessages((prev) => [
          ...prev,
          {
            sender: 'ai',
            text: `「${userMessage}」でお探しですね！楽天トラベルで見つけたおすすめの宿はこちらです：\n\n🏨 ${hotelName}\n🔗 ${hotelUrl}`
          }
        ]);
      } else {
        setMessages((prev) => [
          ...prev,
          { sender: 'ai', text: `「${userMessage}」に一致する宿が見つかりませんでした。別の条件でも試してみてください！` }
        ]);
      }
    } catch (error) {
      setMessages((prev) => [
        ...prev,
        { sender: 'ai', text: 'エラーが発生しました。通信環境を確認してもう一度お試しください。' }
      ]);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={{ maxWidth: '600px', margin: '0 auto', padding: '20px', fontFamily: 'sans-serif' }}>
      <h2 style={{ textAlign: 'center', color: '#ff5500' }}>✈️ 旅行検索AI</h2>
      
      {/* チャット履歴 */}
      <div style={{ border: '1px solid #ccc', borderRadius: '8px', height: '400px', overflowY: 'scroll', padding: '15px', marginBottom: '15px', background: '#f9f9f9' }}>
        {messages.map((msg, index) => (
          <div key={index} style={{ margin: '10px 0', textAlign: msg.sender === 'user' ? 'right' : 'left' }}>
            <span style={{
              display: 'inline-block',
              padding: '10px 14px',
              borderRadius: '12px',
              background: msg.sender === 'user' ? '#0070f3' : '#e5e5ea',
              color: msg.sender === 'user' ? '#fff' : '#000',
              maxWidth: '80%',
              textAlign: 'left',
              whiteSpace: 'pre-wrap'
            }}>
              {msg.text}
            </span>
          </div>
        ))}
        {loading && <div style={{ color: '#888', fontStyle: 'italic' }}>楽天トラベルから検索中...</div>}
      </div>

      {/* 入力フォーム */}
      <form onSubmit={handleSend} style={{ display: 'flex', gap: '10px' }}>
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="例：札幌、温泉、函館など"
          style={{ flex: 1, padding: '10px', borderRadius: '4px', border: '1px solid #ccc' }}
        />
        <button type="submit" style={{ padding: '10px 20px', background: '#ff5500', color: '#fff', border: 'none', borderRadius: '4px', cursor: 'pointer' }}>
          送信
        </button>
      </form>
    </div>
  );
}
